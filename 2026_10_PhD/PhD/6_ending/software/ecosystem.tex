% #1 name  #2 position  #3 role colour  #4 label  #5 kept on the uptake render
\newcommand{\pkg}[5]{%
  \pgfmathsetmacro{\op}{\ifnum\keep=0 1\else\ifnum#5=1 1\else0.22\fi\fi}
  \node[draw=#3,line width=1.3pt,fill=white,rounded corners=6pt,
        minimum width=2.75cm,minimum height=0.95cm,text=ink,
        font=\sffamily\small\bfseries,opacity=\op,text opacity=\op]
       (#1) at #2 {#4};}
